3 · Request proxy (only if needed)

Some sites refuse calls from pages hosted elsewhere. If sign-in or the BGG load fails with a “couldn't be reached” message, deploy the small Cloudflare Worker below (free tier is plenty), paste its URL here, and try again. It only forwards requests to tabletop.events and boardgamegeek.com, and only for your page.

// Cloudflare Worker: forwards only to TTE and BGG, only for your page
const ALLOWED_ORIGIN = 'https://prpost98.github.io'; // change if your Pages URL differs
export default {
  async fetch(req) {
    const cors = {
      'Access-Control-Allow-Origin': ALLOWED_ORIGIN,
      'Access-Control-Allow-Methods': 'GET,POST,OPTIONS',
      'Access-Control-Allow-Headers': 'Authorization,Content-Type',
      'Vary': 'Origin'
    };
    if (req.method === 'OPTIONS') return new Response(null, { headers: cors });
    if (req.headers.get('Origin') !== ALLOWED_ORIGIN) return new Response('Forbidden', { status: 403 });
    const target = new URL(req.url).searchParams.get('url') || '';
    if (!/^https:\/\/(tabletop\.events|boardgamegeek\.com)\//.test(target))
      return new Response('Not allowed', { status: 403, headers: cors });
    const init = { method: req.method, headers: {} };
    const auth = req.headers.get('Authorization'); if (auth) init.headers.Authorization = auth;
    const ct = req.headers.get('Content-Type'); if (ct) init.headers['Content-Type'] = ct;
    if (req.method === 'POST') init.body = await req.text();
    const r = await fetch(target, init);
    const h = new Headers(r.headers);
    for (const [k, v] of Object.entries(cors)) h.set(k, v);
    return new Response(r.body, { status: r.status, headers: h });
  }
};
